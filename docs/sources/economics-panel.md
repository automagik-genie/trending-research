import copy
import hashlib
import json
from pathlib import Path

path = Path("investigations/frontier-economics/sources.json")
raw = path.read_bytes()


def object_without_duplicates(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("duplicate JSON field: " + key)
        result[key] = value
    return result


def nonfinite(value):
    raise ValueError("nonfinite JSON number: " + value)


registry = json.loads(raw, object_pairs_hook=object_without_duplicates,
                      parse_constant=nonfinite)


def need(condition, reason):
    if not condition:
        raise ValueError(reason)


def qualify(data):
    need(data["registry_version"] == "1.0.0", "unsupported registry version")
    rows = data["sources"]
    ids = [row["id"] for row in rows]
    need(len(ids) == len(set(ids)), "duplicate source identity")
    need(set(data["coverage"]["languages"]) == {row["language"] for row in rows},
         "language coverage drift")
    readable = []
    for row in rows:
        need(row["state"] in ("operational", "blocked"), "invalid source state")
        receipt = row["read_receipt"]
        need(receipt["request_url"] == row["url"], "source receipt target drift")
        need(not receipt["retained_original"] and receipt["original_sha256"] is None,
             "uncaptured original promoted")
        need(not row["access"]["recurring_enabled"] and not row["cadence"]["enabled"],
             "unapproved recurring acquisition enabled")
        need(row["rights"]["retention_permission"] is None and
             row["rights"]["redistribution_permission"] is None,
             "unqualified rights grant invented")
        if row["state"] == "operational":
            need(row["blocked"] is None, "blocked source marked operational")
            need(row["access"]["requires_paid_data_access"] is False,
                 "paid source marked operational")
            need(row["kind"] != "platform_preflight", "policy read promoted to data access")
            need(receipt["result"] == "body_read" and receipt["body_verified"] and
                 receipt["response_url"] and receipt["body_markers"],
                 "operational source without body receipt")
            need(row["quota_cost"]["qualification_paid_access_usd"] == 0,
                 "paid receipt marked operational")
            readable.append(row["id"])
        else:
            need(row["blocked"] and row["blocked"]["code"] and
                 row["blocked"]["unblock_requires"], "unnamed blocked source")
    coverage = data["coverage"]
    need(coverage["registered_count"] == len(rows) and
         coverage["one_off_body_qualified_count"] == len(readable) and
         coverage["blocked_count"] == len(rows) - len(readable), "coverage count drift")
    need(coverage["global_sentiment"]["numerator"] is None and
         coverage["global_sentiment"]["denominator"] is None,
         "bounded cohort promoted to global sentiment")
    for gap in coverage["entity_gaps"]:
        need(all(gap[key] is None for key in
                 ("recognized_revenue", "realized_compute_cost",
                  "operating_or_gross_margin", "net_retention")),
             "unknown private metric filled")
    for group in data["origin_groups"]:
        need(set(group["source_ids"]) <= set(ids), "origin references absent source")
        need(group["independent_corroboration_count"] == 0 and
             group["evidence_origin_count"] == 1,
             "common-source evidence counted as independent corroboration")
        for source_id in group["source_ids"]:

            need(group["id"] in source["origin_ids"], "origin mapping drift")
    budget = data["budget"]
    approval = budget["approval"]

    need(budget["actual_model_calls"] == 0 and budget["total_session_cost_usd"] is None,

         "readable candidates confused with release eligibility")
